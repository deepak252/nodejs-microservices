    > aws s3 ls --endpoint-url=<ENDPOINT>       {List all the buckets}
    
    > aws s3 mb s3://posts --endpoint-url=<ENDPOINT>     {/*Make Bucket "posts"*/}

    > aws s3 ls s3://posts --endpoint-url=<ENDPOINT>     {List all files in s3 bucket}
    > aws s3 sync ~/Pictures s3://posts --endpoint-url=<ENDPOINT>     {Upload all files from local Pictures dir to s3 bucket}
